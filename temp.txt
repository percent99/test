/* ============================================================
   1) 공통: updated_at 자동 갱신 트리거
   - 각 테이블의 updated_at을 UPDATE 시 now()로 자동 갱신
   - updated_by는 애플리케이션에서 설정(트리거로 채우려면 세션변수/ROW SECURITY 등 별도 설계 필요)
   ============================================================ */
CREATE OR REPLACE FUNCTION trg_set_updated_at()
RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  NEW.updated_at := now();
  RETURN NEW;
END $$;

/* ============================================================
   2) 코드 관리 (ENUM 대체)
   - code_masters: 코드 마스터(종류). PK: code (텍스트)
   - code_details: 코드 디테일(항목). PK: (master_code, code)
     * 같은 master_code 내에서 code 유일
     * 자연키 기반 FK를 사용해 도메인 무결성 보장
   - 장점: 운영 중 코드 추가/비활성/정렬 변경이 쉬움
   ============================================================ */

DROP TABLE IF EXISTS code_details CASCADE;
DROP TABLE IF EXISTS code_masters CASCADE;

CREATE TABLE code_masters (
  /* 고정 식별자 */
  code           TEXT PRIMARY KEY,                  -- 예: 'user_role', 'mission_status'
  name           TEXT NOT NULL,                     -- 화면/리포트용 이름
  description    TEXT,                              -- 설명
  /* 메타 */
  created_by     UUID,                              -- 생성자(앱 사용자 ID; NULL 허용)
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),-- 생성일시
  updated_by     UUID,                              -- 수정자
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT now() -- 수정일시
);
COMMENT ON TABLE code_masters IS '코드 마스터(ENUM 대체) - 코드 체계의 상위 분류';
COMMENT ON COLUMN code_masters.code IS '마스터 코드(자연키). 예: user_role, mission_status';
COMMENT ON COLUMN code_masters.name IS '마스터 표시명';
COMMENT ON COLUMN code_masters.description IS '마스터 설명';

CREATE TRIGGER code_masters_set_updated_at
BEFORE UPDATE ON code_masters
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

CREATE TABLE code_details (
  /* 복합 자연키: (master_code, code) */
  master_code    TEXT NOT NULL REFERENCES code_masters(code) ON DELETE RESTRICT,
  code           TEXT NOT NULL,                      -- 예: parent / child / approved ...
  name           TEXT NOT NULL,                      -- 화면/리포트용 이름
  description    TEXT,                               -- 항목 설명
  sort_order     INT  NOT NULL DEFAULT 0,            -- 정렬 우선순위
  active         BOOLEAN NOT NULL DEFAULT TRUE,      -- 사용 여부
  /* 보조 식별자(서로게이트 키) */
  id             UUID PRIMARY KEY DEFAULT UUID_V7(), -- 필요 시 조인 최적화/외부연계
  /* 메타 */
  created_by     UUID,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by     UUID,
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (master_code, code)
);
COMMENT ON TABLE code_details IS '코드 디테일(ENUM 항목 대체). (master_code, code) 복합 PK';
COMMENT ON COLUMN code_details.master_code IS '소속 마스터 코드';
COMMENT ON COLUMN code_details.code IS '디테일 코드(자연키). 예: approved, child';
COMMENT ON COLUMN code_details.id IS '서로게이트 UUID 키(선택 사용)';

CREATE INDEX idx_code_details_master_active ON code_details(master_code, active, sort_order);

CREATE TRIGGER code_details_set_updated_at
BEFORE UPDATE ON code_details
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* ============================================================
   3) 도메인 테이블
   - 모든 테이블 공통: id(UUID v7), created_by/at, updated_by/at
   - 코드 컬럼은 (고정 마스터 생성 칼럼 + 디테일 코드) 복합FK 패턴 사용
     예) role_master_code = 'user_role' (생성 칼럼, 자동 저장)
         role_code        = 'parent'   (입력값)
         FK(role_master_code, role_code) → code_details(master_code, code)
   ============================================================ */

DROP TABLE IF EXISTS redemptions CASCADE;
DROP TABLE IF EXISTS rewards CASCADE;
DROP TABLE IF EXISTS points_ledger CASCADE;
DROP TABLE IF EXISTS mission_proofs CASCADE;
DROP TABLE IF EXISTS missions CASCADE;
DROP TABLE IF EXISTS mission_templates CASCADE;
DROP TABLE IF EXISTS parent_child CASCADE;
DROP TABLE IF EXISTS users CASCADE;
DROP TABLE IF EXISTS households CASCADE;

/* 3.1 가정(가족) */
CREATE TABLE households (
  id           UUID PRIMARY KEY DEFAULT UUID_V7(),        -- PK (시간정렬 우수)
  name         TEXT NOT NULL,                              -- 가족명
  /* 메타 */
  created_by   UUID,                                       -- 생성자(앱 사용자)
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),         -- 생성일시
  updated_by   UUID,                                       -- 수정자
  updated_at   TIMESTAMPTZ NOT NULL DEFAULT now()          -- 수정일시
);
COMMENT ON TABLE households IS '가족(하우스홀드) 단위 관리 테이블';

CREATE TRIGGER households_set_updated_at
BEFORE UPDATE ON households
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.2 사용자 (부모/자녀) */
CREATE TABLE users (
  id                 UUID PRIMARY KEY DEFAULT UUID_V7(),   -- PK
  household_id       UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE, -- 소속 가족
  /* 역할 코드: user_role */
  role_master_code   TEXT GENERATED ALWAYS AS ('user_role') STORED, -- 고정 마스터
  role_code          TEXT NOT NULL,                                  -- 디테일 코드(parent/child)
  CONSTRAINT fk_users_role
    FOREIGN KEY (role_master_code, role_code)
    REFERENCES code_details(master_code, code)
    ON UPDATE RESTRICT ON DELETE RESTRICT,
  display_name       TEXT NOT NULL,                          -- 표시명(닉네임)
  birthdate          DATE,                                   -- 생일(선택)
  email              TEXT,                                   -- 이메일(선택, 고유 제약은 필요 시)
  UNIQUE (household_id, display_name),                       -- 가족 내 표시명 중복 방지(권장)
  /* 메타 */
  created_by         UUID,
  created_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by         UUID,
  updated_at         TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE users IS '가족 구성원(부모/자녀). 역할은 코드테이블로 관리';

CREATE INDEX idx_users_household ON users(household_id);
CREATE TRIGGER users_set_updated_at
BEFORE UPDATE ON users
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.3 (선택) 부모-자녀 관계 명시 */
CREATE TABLE parent_child (
  parent_id      UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE, -- 부모
  child_id       UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE, -- 자녀
  PRIMARY KEY (parent_id, child_id),
  /* 메타 */
  created_by     UUID,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by     UUID,
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE parent_child IS '부모-자녀 매핑(권한 구분/알림 라우팅 등)';

CREATE TRIGGER parent_child_set_updated_at
BEFORE UPDATE ON parent_child
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.4 미션 템플릿(반복 규칙 포함) */
CREATE TABLE mission_templates (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  household_id     UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
  title            TEXT NOT NULL,                              -- 템플릿 제목
  description      TEXT,                                       -- 설명
  default_points   INT  NOT NULL CHECK (default_points >= 0),  -- 기본 포인트
  repeat_rule      TEXT,                                       -- iCal RRULE 등 (예: FREQ=DAILY)
  active           BOOLEAN NOT NULL DEFAULT TRUE,              -- 사용 여부
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE mission_templates IS '미션 템플릿. 반복 미션 스케줄 생성의 기준';

CREATE INDEX idx_mtpl_household_active ON mission_templates(household_id, active);
CREATE TRIGGER mission_templates_set_updated_at
BEFORE UPDATE ON mission_templates
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.5 미션(인스턴스) */
CREATE TABLE missions (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  household_id     UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
  template_id      UUID REFERENCES mission_templates(id),          -- 출처 템플릿(선택)
  child_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  title            TEXT NOT NULL,                                  -- 제목(스냅샷/자유입력)
  description      TEXT,
  points           INT  NOT NULL CHECK (points >= 0),              -- 적립 포인트
  due_at           TIMESTAMPTZ,                                    -- 마감일시
  /* 상태 코드: mission_status */
  status_master_code TEXT GENERATED ALWAYS AS ('mission_status') STORED,
  status_code        TEXT NOT NULL,
  CONSTRAINT fk_missions_status
    FOREIGN KEY (status_master_code, status_code)
    REFERENCES code_details(master_code, code)
    ON UPDATE RESTRICT ON DELETE RESTRICT,
  submitted_at     TIMESTAMPTZ,                                    -- 자녀 제출 시각
  approved_at      TIMESTAMPTZ,                                    -- 승인 시각
  approved_by      UUID REFERENCES users(id),                       -- 승인자(부모)
  cancelled_at     TIMESTAMPTZ,                                    -- 취소 시각
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  /* 간단 무결성: approved면 approved_at 있어야 함 */
  CONSTRAINT approved_needs_timestamp CHECK (
    status_code <> 'approved' OR approved_at IS NOT NULL
  )
);
COMMENT ON TABLE missions IS '할당된 개별 미션(자녀별). 상태는 코드테이블 참조';

CREATE INDEX idx_missions_child_status_due ON missions(child_id, status_code, due_at);
CREATE INDEX idx_missions_household_created ON missions(household_id, created_at DESC);
CREATE TRIGGER missions_set_updated_at
BEFORE UPDATE ON missions
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.6 미션 증빙(사진/메모 등) */
CREATE TABLE mission_proofs (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  mission_id       UUID NOT NULL REFERENCES missions(id) ON DELETE CASCADE,
  /* 증빙 종류: proof_kind */

  url              TEXT,                             -- 파일 저장소 URL 또는 키
  note             TEXT,                             -- 메모
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE mission_proofs IS '미션 증빙 자료(여러 개 가능): 사진, 메모 등';

CREATE INDEX idx_mproofs_mission ON mission_proofs(mission_id, created_at);
CREATE TRIGGER mission_proofs_set_updated_at

FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.7 포인트 원장 (단일 진실 원장) */
CREATE TABLE points_ledger (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  household_id     UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
  child_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  /* 원장 출처: ledger_source */
  source_master_code TEXT GENERATED ALWAYS AS ('ledger_source') STORED,
  source_code        TEXT NOT NULL,
  CONSTRAINT fk_ledger_source
    FOREIGN KEY (source_master_code, source_code)
    REFERENCES code_details(master_code, code)
    ON UPDATE RESTRICT ON DELETE RESTRICT,
  source_id        UUID,                                -- 예: missions.id, redemptions.id
  amount           INT  NOT NULL CHECK (amount <> 0),   -- 적립:+ / 차감:-
  description      TEXT,
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE points_ledger IS '포인트 적립/차감의 유일 원장. 합계가 잔액';

CREATE INDEX idx_ledger_child_time ON points_ledger(child_id, created_at);
CREATE INDEX idx_ledger_source ON points_ledger(source_code, source_id);
CREATE TRIGGER points_ledger_set_updated_at
BEFORE UPDATE ON points_ledger
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.8 보상(상품) */
CREATE TABLE rewards (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  household_id     UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
  name             TEXT NOT NULL,                          -- 상품명
  description      TEXT,
  points_cost      INT  NOT NULL CHECK (points_cost > 0),  -- 필요 포인트
  stock            INT  CHECK (stock IS NULL OR stock >= 0),-- NULL=무제한
  active           BOOLEAN NOT NULL DEFAULT TRUE,          -- 사용 여부
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE rewards IS '교환 가능한 보상(상품). 비용/재고/활성 여부 포함';

CREATE INDEX idx_rewards_household_active ON rewards(household_id, active);
CREATE TRIGGER rewards_set_updated_at
BEFORE UPDATE ON rewards
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* 3.9 보상 교환 */
CREATE TABLE redemptions (
  id               UUID PRIMARY KEY DEFAULT UUID_V7(),
  household_id     UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
  child_id         UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  reward_id        UUID NOT NULL REFERENCES rewards(id),
  /* 교환 상태: redemption_status */
  status_master_code TEXT GENERATED ALWAYS AS ('redemption_status') STORED,
  status_code        TEXT NOT NULL,
  CONSTRAINT fk_redemptions_status
    FOREIGN KEY (status_master_code, status_code)
    REFERENCES code_details(master_code, code)
    ON UPDATE RESTRICT ON DELETE RESTRICT,
  points_cost_snapshot INT NOT NULL CHECK (points_cost_snapshot > 0), -- 당시 비용 스냅샷
  approved_by      UUID REFERENCES users(id),                          -- 승인자(부모)
  approved_at      TIMESTAMPTZ,
  fulfilled_at     TIMESTAMPTZ,
  cancelled_at     TIMESTAMPTZ,
  created_by       UUID,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_by       UUID,
  updated_at       TIMESTAMPTZ NOT NULL DEFAULT now()
);
COMMENT ON TABLE redemptions IS '상품 교환 요청/승인/지급/취소 내역. 차감은 원장에서 처리';

CREATE INDEX idx_redemptions_child_time ON redemptions(child_id, created_at DESC);
CREATE TRIGGER redemptions_set_updated_at
BEFORE UPDATE ON redemptions
FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

/* ============================================================
   4) 뷰/유틸
   - v_child_points_balance: 자녀별 잔액 조회
   ============================================================ */

CREATE OR REPLACE VIEW v_child_points_balance AS
SELECT
  u.id            AS child_id,
  u.display_name,
  u.household_id,
  COALESCE(SUM(l.amount), 0) AS balance
FROM users u
LEFT JOIN points_ledger l ON l.child_id = u.id
WHERE (u.role_master_code, u.role_code) = ('user_role','child')
GROUP BY u.id, u.display_name, u.household_id;

COMMENT ON VIEW v_child_points_balance IS '자녀별 포인트 잔액(원장 합계 기반)';
